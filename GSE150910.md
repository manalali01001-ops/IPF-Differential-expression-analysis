***Log2 fold change***
epsilon is just a number added in the equation of log2 folds to prevent values of 0 of creating and error


```python
epsilon = 1
control_mean = np.mean(control)
ipf_mean = np.mean(ipf)

FC = (ipf_mean + epsilon) / (control_mean + epsilon)
log2_FC = np.log2(FC)

print(log2_FC)
```

    0.7894289821093974



```python
p_values = []
log2_FCs = []

epsilon = 1


for i in range(data.shape[0]):
    control = data[i, labels == 'control']
    ipf = data[i, labels == 'ipf']

    ks_statistics, p_value = ks_2samp(control, ipf)
    p_values.append(p_value)

    control_mean = np.mean(control)
    ipf_mean = np.mean(ipf)

    FC = (ipf_mean + epsilon) / (control_mean + epsilon)
    log2_FC = np.log2(FC)
    log2_FCs.append(log2_FC)

p_values = np.array(p_values)
log2_FCs = np.array(log2_FCs)
```


```python
p_values_bonf = p_values * len(genes) #Bonbonniere correction
```

```python
epsilon = 1

for i in range(len(sig_genes)):
    control_mean = np.mean(sig_data[i, labels == 'control'])

    sig_data_log2fc[i, :] = np.log2((sig_data[i,:] + epsilon) / (control_mean + epsilon))
```


```python
row_linkage = fs.linkage(sig_data_log2fc, method='average')
col_linkage = fs.linkage(sig_data_log2fc.T, method='average')

sns_plot = sns.clustermap(
    sig_data_log2fc,
    row_linkage=row_linkage,
    col_linkage=col_linkage,
    xticklabels=samples,
    yticklabels=sig_genes,
    cmap='coolwarm',
    vmin=-6,
    vmax=6
)

sns_plot.ax_heatmap.set_xticklabels(
```
